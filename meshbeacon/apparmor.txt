#include <tunables/global>

# MeshBeacon AppArmor profile (architecture §4.7). Based on the Home Assistant example app profile.
# STATUS: not yet validated on a real AppArmor-enabled host — it must be checked on the reference
# Raspberry Pi in complain mode before release (roadmap LP-2 / RG-2). See DOCS.md.
profile meshbeacon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities used by the root init step only (ownership and permissions of /data).
  capability chown,
  capability fowner,
  capability dac_override,
  capability setuid,
  capability setgid,
  capability kill,

  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /etc/** r,
  /tmp/** rwk,

  # Python runtime and the MeshBeacon core (read-only).
  /usr/local/bin/python3* ix,
  /usr/local/lib/** mr,
  /usr/lib/** mr,
  /usr/share/** r,

  # App data: options, identity, Reticulum state.
  /data/ r,
  /data/** rwk,

  # Process information used by the resource guards.
  @{PROC}/** r,
  /sys/fs/cgroup/** r,

  # Outbound TCP to user-configured peers, DNS lookups, and the internal health endpoint.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,

  deny /data/options.json w,
  deny mount,
  deny ptrace,
}
