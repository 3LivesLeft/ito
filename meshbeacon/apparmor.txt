#include <tunables/global>

# MeshBeacon AppArmor profile.
#
# The Supervisor renames the profile to the app's slug (for a local app: local_meshbeacon) and
# loads it only when the app is INSTALLED or UPDATED to a new version — not on Rebuild (verified in
# Supervisor 2026.11.0). See docs/install-local-prototype.md for updating an existing install.
#
# Design (architecture §4.7), revised after the v0.1.0 startup failure on the reference Pi:
# - Read access to the container's own filesystem is allowed. s6-overlay starts with shell and
#   execline SCRIPTS (/init is '#!/bin/sh -e'), and an interpreter must be able to READ a script
#   it executes; the previous profile granted only 'ix' on them, so /bin/sh could not open /init.
# - Execution is allowed only from the image's program directories and s6-overlay's run-time
#   service directories — never from /data or /tmp.
# - Writes are allowed only to /data (app data), s6-overlay's own folders in /run, the discovery
#   relay's folder (its owner only) and /tmp (M1.2: /run narrowed from a blanket rule).
# - Only the capabilities the root init step and s6-overlay need; mount and ptrace are denied.
#
# Enforcement is tested in CI on an AppArmor-enabled runner (tests/container, MESHBEACON_APPARMOR).
profile meshbeacon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities (reviewed M1.2; none added): root init step and relay folder ownership (chown,
  # fowner, dac_override on /data after it is handed to the service user), s6-overlay dropping to
  # the service user (setuid, setgid) and signalling its children (kill). The discovery relay
  # itself needs none of them beyond being root. The unprivileged core and Reticulum cannot use
  # any capability: they run as an ordinary user without file capabilities (no-new-privileges).
  capability chown,
  capability fowner,
  capability dac_override,
  capability setuid,
  capability setgid,
  capability kill,
  # Not granted, and denied explicitly so that it is not logged: root's chmod of a file whose group
  # it is not in (the init step, after handing /data to the service user) makes the kernel check
  # fsetid only to decide whether to keep a setgid bit, which MeshBeacon never sets. Without this
  # rule every start logged three harmless denials (seen in CI under enforcement).
  deny capability fsetid,

  # Signals within the container (s6 supervision tree) and from the container runtime (stop).
  signal (send, receive) peer=@{profile_name},
  signal (receive) peer=unconfined,

  # Read the container's own filesystem (scripts must be readable to be interpreted).
  / r,
  /** r,

  # Execute: s6-overlay, the shell and system tools, Python, and MeshBeacon's s6 scripts.
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** ix,
  /etc/cont-init.d/** ix,
  /etc/cont-finish.d/** ix,
  /etc/services.d/** ix,
  /run/{s6,s6-rc*,service}/** ix,

  # Shared libraries and Python extension modules.
  /lib/** mr,
  /usr/lib/** mr,
  /usr/local/lib/** mr,

  # Writable locations.
  /data/ rw,
  /data/** rwk,
  /tmp/{,**} rwk,

  # /run, as s6-overlay 3.2.3.2 uses it (libexec/preinit, s6-linux-init, s6-rc-init -c /run/s6/db
  # /run/service): it checks that /run is writable, then keeps its state in these folders.
  # Nothing else in /run is writable, so neither the core nor Reticulum can create files there.
  # s6-linux-init sets the owner and mode of /run itself ('w' on the folder is that change only;
  # creating anything inside is checked against the rules below, seen under enforcement in CI).
  /run/ rw,
  "/run/test of writability" rw,
  /run/s6/{,**} rwk,
  /run/s6-rc* rwk,
  /run/s6-rc*/{,**} rwk,
  /run/service/{,**} rwk,
  /run/uncaught-logs/{,**} rwk,
  /run/s6-linux-init-container-results/{,**} rwk,
  # The discovery relay's folder and socket: root's alone (owner rule on top of 0700/0600), so the
  # service user, which Reticulum runs as, cannot reach them even if file modes were wrong.
  owner /run/meshbeacon/{,**} rwk,
  # Channels between the core and its services (ADR 0018): FIFOs and the prepare mode file. File
  # modes (root:meshbeacon 0660 in 0750 folders) keep Reticulum's user out; nothing here runs.
  /run/meshbeacon-io/{,**} rw,
  # The optional messaging engine's socket and the settings the core writes for it (M4, D-M4-8;
  # created only when messaging is on). File modes (mb-lxmf:mb-msg and meshbeacon:mb-msg, 0750)
  # keep Reticulum's user out; nothing here runs. Validated under enforcement in CI since M4 WP6.
  /run/meshbeacon-messaging/{,**} rw,
  # The shared instance's RPC key while messaging is on (ADR 0023): written by root's init step,
  # read by Reticulum's user and the engine's user through the group mb-rpc (0750/0640).
  /run/meshbeacon-rpc/{,**} rw,
  # Nomad Network pages' socket (v0.6, ADR 0027; created only when pages are on). File modes
  # (mb-nomad:mb-page 0750) keep Reticulum's and messaging's users out; nothing here runs. The
  # published pages in /data/pages are covered by the /data rule above and, like all of /data,
  # can never be executed: Nomad Network runs executable pages as scripts, and AppArmor refuses.
  /run/meshbeacon-pages/{,**} rw,
  /dev/tty rw,

  # The Supervisor's per-app socket (newer Supervisors; used only by the discovery relay).
  # AppArmor checks a Unix socket by the name the server bound it to, which is the Supervisor's
  # own path, not /run/supervisor/api/supervisor.sock as mounted here (Supervisor e610ca8:
  # config.py path_run_supervisor = /run/os/supervisor, apps/app.py apps/<slug>); the mounted name
  # is listed too. The socket is root's (0600), and only the relay, which runs as root, connects.
  owner /run/os/supervisor/apps/*/supervisor.sock rw,
  owner /run/supervisor/api/supervisor.sock rw,

  # Outbound TCP to user-configured peers, DNS lookups, the internal health endpoint, and the
  # local sockets and pipes used by s6-overlay.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,

  deny mount,
  deny umount,
  deny pivot_root,
  deny ptrace (trace),
}
